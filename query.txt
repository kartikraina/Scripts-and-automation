  CREATE OR REPLACE EDITIONABLE FUNCTION "DBAHC"."GE_PASSWORD_VERIFY_FUNCTION" 
(username varchar2, password varchar2, old_password varchar2)
RETURN boolean IS
-- --------------------------------------------------------------------
-- Tony Clark
-- April 2013
--
-- Must be a minimum of 8 characters in length
-- Must be a maximum of 30 characters long.
-- Must contain a minimum of 2 embedded numbers.
-- Must begin and end with an alphabet character.
-- Shall not be the same as the User ID
-- Shall not be a known word, only few simple words are checked.
-- Special characters are allowed, but not required ( # _ $ ).
-- In 11g and above, shall contain at least to one uppercase and one lowercase letter.
-- Password cannot be reused
-- --------------------------------------------------------------
n boolean;
m integer;
p integer;
digitcount number;
ischar  boolean;
isspec boolean;
digitarray varchar2(250);
chararray varchar2(250);
upperarray varchar2(250);
specarray varchar2(25);
username_lower varchar2(250);
password_lower varchar2(250);
pwd varchar2(250);
BEGIN
digitarray:= '0123456789';
chararray:= 'abcdefghijklmnopqrstuvwxyz';
upperarray:= 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
specarray:= '#_$';

-- Check if the password is same or includes the username

IF NLS_LOWER(password) = NLS_LOWER(username) THEN
  raise_application_error(-20001, 'Password same as or similar to username');
END IF;
pwd:=password;
username_lower := NLS_LOWER(username);
password_lower := NLS_LOWER(password);
m := length(password);
p := length(username);
FOR i in 1..m loop
  if substr(password_lower,i,1) = substr(username_lower,1,1) then
    if substr(password_lower,i,p) = substr(username_lower,1,p) then
     --dbms_output.put_line('a');
     raise_application_error(-20001, 'Username is contained in the password');
    end if;
  end if;
end loop;

-- Check for the minimum length of the password
IF length(password) < 8 THEN
   raise_application_error(-20002, 'Password length less than 8');
END IF;

-- Check for the maximum length of the password
IF length(password) > 30 THEN
   raise_application_error(-20002, 'Password length more than 30');
END IF;

-- Check if the password is too simple. A dictionary of words may be
-- maintained and a check may be made so as not to allow the words
-- that are too simple for the password.
IF NLS_LOWER(password) IN ('welcome', 'database', 'account', 'user', 'password', 'oracle', 'computer', 'abcd') THEN
   raise_application_error(-20002, 'Password too simple');
END IF;

--Check for the lowercase character
ischar:=FALSE;
FOR i IN 1..length(chararray) LOOP
   FOR j IN 1..m LOOP
      IF substr(password,j,1) = substr(chararray,i,1) THEN
         ischar:=TRUE;
          GOTO findupperchar;
      END IF;
   END LOOP;
END LOOP;
IF ischar = FALSE THEN
   raise_application_error(-20003, 'Password should contain at least one lowercase letter');
END IF;

<<findupperchar>>
--Check for the uppercase character
ischar:=FALSE;
FOR i IN 1..length(upperarray) LOOP
   FOR j IN 1..m LOOP
      IF substr(password,j,1) = substr(upperarray,i,1) THEN
         ischar:=TRUE;
          GOTO findembednum;
      END IF;
   END LOOP;
END LOOP;
IF ischar = FALSE THEN
   raise_application_error(-20004, 'Password should contain at least one uppercase letter');
END IF;

<<findembednum>>
--Check for the embedded numbers

FOR i in 1..10 loop
IF SUBSTR(password,1,1) = SUBSTR(digitarray,i,1) OR substr(password,m,1) = substr(digitarray,i,1) then
raise_application_error(-20004, 'Password cannot start or end with number');
end if;
end loop;

digitcount := 0;
FOR i in 1..10 loop
 FOR j in 2..m-1 loop
 IF substr(password,j,1) = substr(digitarray,i,1) then
  digitcount := digitcount+1;
 end if;
 end loop;
end loop;
IF digitcount < 2 then
raise_application_error(-20005, 'Password should contain minimum two embedded numbers');
end if;

 -- Everything is fine; return TRUE ;
RETURN(TRUE);
END;

-- Datafile resize 

set linesize 1000 pagesize 0 feedback off trimspool on
with
hwm as (
  -- get highest block id from each datafiles ( from x$ktfbue as we don't need all joins from dba_extents )
  select /*+ materialize */ ktfbuesegtsn ts#,ktfbuefno relative_fno,max(ktfbuebno+ktfbueblks-1) hwm_blocks
  from sys.x$ktfbue group by ktfbuefno,ktfbuesegtsn
),
hwmts as (
  -- join ts# with tablespace_name
  select name tablespace_name,relative_fno,hwm_blocks
  from hwm join v$tablespace using(ts#)
),
hwmdf as (
  -- join with datafiles, put 5M minimum for datafiles with no extents
  select file_name,nvl(hwm_blocks*(bytes/blocks),5*1024*1024) hwm_bytes,bytes,autoextensible,maxbytes
  from hwmts right join dba_data_files using(tablespace_name,relative_fno)
)
select
case when autoextensible='YES' and maxbytes>=bytes
then -- we generate resize statements only if autoextensible can grow back to current size
  '/* reclaim '||to_char(ceil((bytes-hwm_bytes)/1024/1024),999999)
   ||'M from '||to_char(ceil(bytes/1024/1024),999999)||'M */ '
   ||'alter database datafile '''||file_name||''' resize '||ceil(hwm_bytes/1024/1024)||'M;'
else -- generate only a comment when autoextensible is off
  '/* reclaim '||to_char(ceil((bytes-hwm_bytes)/1024/1024),999999)
   ||'M from '||to_char(ceil(bytes/1024/1024),999999)
   ||'M after setting autoextensible maxsize higher than current size for file '
   || file_name||' */'
end SQL
from hwmdf
where
bytes-hwm_bytes>1024*1024*1024 -- resize only if at least 1MB can be reclaimed
order by bytes-hwm_bytes desc;

-- Tablespace-level capacity and free space (Permanent tablespaces)
SELECT
    dfs.tablespace_name,
    ROUND(SUM(dfs.bytes)/1024/1024, 2) AS total_mb,
    ROUND(SUM(dfs.bytes - NVL(free.free_bytes, 0))/1024/1024, 2) AS used_mb,
    ROUND(SUM(NVL(free.free_bytes, 0))/1024/1024, 2) AS free_mb,
    ROUND((SUM(NVL(free.free_bytes, 0))/SUM(dfs.bytes))*100, 2) AS pct_free
FROM (
    SELECT tablespace_name, file_id, bytes FROM dba_data_files
) dfs
LEFT JOIN (
    SELECT tablespace_name, file_id, SUM(bytes) AS free_bytes
    FROM dba_free_space
    GROUP BY tablespace_name, file_id
) free
ON free.tablespace_name = dfs.tablespace_name
AND free.file_id = dfs.file_id
GROUP BY dfs.tablespace_name
ORDER BY dfs.tablespace_name;


-- HWM per datafile: highest allocated block number
-- Requires DBA privileges
SELECT
  de.file_id,
  ddf.file_name,
  ddf.tablespace_name,
  MAX(de.block_id + de.blocks) AS hwm_block
FROM dba_extents de
JOIN dba_data_files ddf
  ON ddf.file_id = de.file_id
WHERE ddf.tablespace_name='EPOP3_ORACLE'
GROUP BY de.file_id, ddf.file_name, ddf.tablespace_name
ORDER BY de.file_id;


WITH hwm AS (
  SELECT file_id, MAX(block_id + blocks) AS hwm_block
  FROM dba_extents
  GROUP BY file_id
),
bs AS (
  SELECT value AS block_size FROM v$parameter WHERE name = 'db_block_size'
)
SELECT
  d.file_id,
  d.file_name,
  ROUND(d.bytes/1024/1024, 2) AS current_size_mb,
  ROUND(NVL(h.hwm_block, 1) * (SELECT block_size FROM bs) / 1024 / 1024, 2) AS hwm_size_mb,
  ROUND((NVL(h.hwm_block, 1) * (SELECT block_size FROM bs) + 64*1024*1024) / 1024 / 1024, 2) AS suggested_resize_mb
FROM dba_data_files d
LEFT JOIN hwm h ON h.file_id = d.file_id
ORDER BY d.file_id;

# Total number of DML executions by a user


SELECT
  TRUNC(sn.begin_interval_time) AS day,
  c.command_name,
  SUM(ss.executions_delta) AS dml_exec
FROM
  dba_hist_sqlstat ss
  JOIN dba_hist_snapshot sn
    ON sn.dbid = ss.dbid
   AND sn.instance_number = ss.instance_number
   AND sn.snap_id = ss.snap_id
  JOIN v$sql sq
    ON sq.sql_id = ss.sql_id
  JOIN v$sqlcommand c
    ON c.command_type = sq.command_type
WHERE
  sn.begin_interval_time >= SYSDATE - 7
  AND ss.parsing_user_id = (SELECT user_id FROM dba_users WHERE username = :username)
  AND c.command_name IN ('INSERT','UPDATE','DELETE','MERGE')
GROUP BY
  TRUNC(sn.begin_interval_time),c.command_name
ORDER BY
  day,command_name;
