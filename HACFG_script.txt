OCR_DISK_LIST=''

#Labelling the OCR disks...

#Check if AFD is installed or not...

if [[ `nohup $CRS_HOME/bin/asmcmd afd_state | grep -i ASMCMD-9526` == '' ]]
then
AFD_STATE=0
else
AFD_STATE=1
fi

#Label Disks ....

for disk in `cat $OCR_DISK_LIST_PATH`;
do
MPATH_NAME=`echo ${disk} | awk -F '/' '{print $NF}'`
AFD_LABEL=${OCR_DG_PREFIX}_${MPATH_NAME}
#echo "$AFD_LABEL"
if [[ `blkid | grep TYPE="oracleasm" | grep $disk` == '' && `blkid | grep $disk` == '' ]]
then
echo Disk $disk not being used by ASM.
echo "Labelling ocr disk $disk now..."
export ORACLE_BASE=/tmp
if [[ $AFD_STATE == 0 ]]
then
echo "$CRS_HOME/bin/asmcmd afd_label $AFD_LABEL $disk --init"
$CRS_HOME/bin/asmcmd afd_label ${AFD_LABEL} ${disk} --init
else
echo "$CRS_HOME/bin/asmcmd afd_label $AFD_LABEL $disk"
$CRS_HOME/bin/asmcmd afd_label ${AFD_LABEL} ${disk}
fi
echo "OCR Disk $disk labelled as --> $AFD_LABEL"
chown grid:asmadmin $disk
if [[ $OCR_DISK_LIST == '' ]]; then
OCR_DISK_LIST=$disk
else
OCR_DISK_LIST=$OCR_DISK_LIST,${disk}
fi
unset ORACLE_BASE
else
echo Disk $disk is being used by ASM already.
echo skipping disk $disk..
fi
done;
